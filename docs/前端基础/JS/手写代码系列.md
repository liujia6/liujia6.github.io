### [call 模拟](https://github.com/mqyqingfeng/Blog/issues/11)

f.call(obj,args1,args2)

1. 如何获取调用函数的函数呢
	1. 函数内的 this 就是调用该方法的函数，this.指向调用该方法的对象
2. 如何模拟函数执行并修改 this 呢
	1. 思考在 JS 中有哪几种操作会改变 this？（硬绑定通过 new 一个对象，软绑定通过对象嵌套函数调用，bind 方法）
	2. 对了！我们可以用软绑定通过对象嵌套修改函数的 this（具体可以查看 JS 章节的基础内容对 this 的介绍）
	3. 通过给对象加一个属性为这个函数，再通过这个对象调用此函数即可把函数的 this 改为这个对象

```js
Function.prototype.call = function (context = window, ...arg) {
  context.fn = this;
  const result = context.fn(...args);
  delete context.fn;
  return result;
};
```

### [bind 模拟](https://github.com/mqyqingfeng/Blog/issues/12)

```js
const func = aFn.bind(obj)

Function.prototype.myBind = function (bindCtx, ...preArgs) {
  const originFn = this;
  if (typeof originFn !== 'function') {
    throw new TypeError('myBind 只能由函数调用');
  }

  const boundFn = function (...callArgs) {
    // 判断是否 new 调用
    const realThis = this instanceof boundFn ? this : bindCtx;
    return originFn.apply(realThis, [...preArgs, ...callArgs]);
  };

  // 核心：用 setPrototypeOf 做原型隔离，代替 new Temp()
  boundFn.prototype = Object.create(originFn.prototype);
  // 等价 boundFn.prototype = new Temp();
  // 也可以拆分写成两行：
  // const proto = Object.create(originFn.prototype);
  // Object.setPrototypeOf(boundFn, proto);

  return boundFn;
};
```

- bind() 方法会创建一个新函数。当这个新函数被调用时，bind() 的第一个参数将作为它运行时的 this，之后的一序列参数将会在传递的实参前传入作为它的参数
- 一个绑定函数也能使用 new 操作符创建对象：这种行为就像把原函数当成构造器。提供的 this 值被忽略，同时调用时的参数被提供给模拟函数。

```javascript

Function.prototype.myBind = function (context, ...args1) {
  const _this = this;
  return function F(...args2) {
    // 判断是否用于构造函数
    if (this instanceof F) {
      return new _this(...args1, ...args2);
    }
    return _this.apply(context, args1.concat(args2));
  };
};
```

### [防抖和节流的应用区分考虑](https://vue3js.cn/interview/JavaScript/debounce_throttle.html#%E4%B8%80%E3%80%81%E6%98%AF%E4%BB%80%E4%B9%88)

都是在一段时间内用户高频触发某个事件

- 防抖指的是，我们高频触发某个事件的时候，该事件只需要在多次触发后执行最终有效的一次。
	- 只需要在用户操作完执行一次的操作
- 而节流指的是，某个事件在一段时间内高频触发，但它又需要及时去响应多次执行。我们可以通过节流减少它在一段时间内的触发频率，从而提高性能，保证它在一段时间内触发一次。
	- 节流的事件是需要及时给用户响应事件的，只是减少太高的频率调用

### 函数节流 (触发后规定时间内忽略)

鼠标不断点击触发，点击事件在规定时间内只触发一次 (单位时间内只触发一次)

监听滚动事件，比如是否滑到底部自动加载更多，用 throttle 来判断

```javascript
// 延迟触发
function throttleTimer(fn, ms) {
  let timer = null;
  return (...args) => {
    if (timer) return;
    timer = setTimeout(() => {
      fn.apply(this, args);
      timer = null;
    }, ms);
  };
}

// 立即触发
function throttleDate(fn, ms) {
  let pre = 0;
  let that = this;
  return function (...args) {
    let now = Date.now();
    if (now - pre > ms) {
      fn.apply(that, args);
      pre = now;
    }
  };
}

// 首尾都能触发
function throttleAll(fn, ms) {
  let timer = null;
  let that = this;
  let pre = 0;
  return function () {
    let now = Date.now();
    let that = this;
    let remaining = delay - (now - pre);
    if (timer) clearTimeout(timer);
    if (remaining <= 0) {
      fn.apply(that, arguments);
      pre = Date.now();
    } else {
      timer = setTimeout(() => {
        fn.apply(this, arguments);
      }, remaining);
    }
  };
}
```

### 深度克隆

1. 原始类型直接返回
2. 判断是数组 / 对象，创建对应空容器
3. 遍历属性，递归深克隆
4. 返回新对象

```javascript
function clone(target, map = new weakMap()) {
  if (type == null || typeof target !== "object") {
    return target;
  } else {
    let cloneTarget = Array.isArray(target) ? [] : {};
    if (map.has(target)) {
      return target;
    }
    map.set(target, cloneTarget);
    for (let key in target) {
      cloneTarget[key] = clone(target[key], map);
    }
    return cloneTarget;
  }
}
```

```javascript
Promise.myAll = function (promises) {
  // 1. 返回一个新的 Promise
  return new Promise((resolve, reject) => {
    // 转成数组（兼容类数组）
    const arr = Array.from(promises);
    const len = arr.length;
    const result = [];
    let count = 0;

    // 空数组直接 resolve
    if (len === 0) {
      return resolve([]);
    }

    arr.forEach((item, index) => {
      // 把每一项都转成 Promise（非 promise 直接包装）
      Promise.resolve(item)
        .then((res) => {
          result[index] = res; // 按顺序存结果
          count++; // 计数

          // 全部完成 → 返回结果数组
          if (count === len) {
            resolve(result);
          }
        })
        .catch((err) => {
          // 失败直接拒绝，短路效应
          reject(err);
        });
    });
  });
};
```

### 实现事件类 Event

包括 on()，off()，once()，emit() 方法

```javascript
class Event {
  constructor() {
    this.events =  new Map();
  }
  on(event, callback) {
    if (!this.events.get(event)) {
      this.events.set(event,[])
    }
    this.events[event].push(callback);
  }
  off(event, callback) {
    if (this.events[event]) {
      this.events[event] = this.events[event].filter((fn) => callback !== fn);
    }
  }
  //为指定事件注册一个单次监听器，即 监听器最多只会触发一次，触发后立刻解除该监听器。
  once(event, callback) {
    let _this = this;
    if (!_this.events[event]) {
      _this.events[event] = [];
    }
    const fn = function () {
      callback.apply(_this, arguments);
      _this.off(event, fn);
    };
    _this.events[event].push(fn);
  }
  emit(event, data) {
    if (this.events[event]) {
      this.events[event].forEach(function (callback) {
        callback(data);
      });
    }
  }
}

let event1 = new Event();

event1.emit("abc", 521);
event1.emit("abc", 520);
```
